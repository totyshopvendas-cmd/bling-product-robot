@echo off
title TotyShop - Instalar atualizacao
set "ZIP=%TEMP%\t.zip"
set "URL=https://github.com/totyshopvendas-cmd/bling-product-robot/raw/cfafaf5/atualizacao_robo_dropper_planilha.zip"
echo Procurando a pasta do projeto...
set "PROJ="
for /f "delims=" %%F in ('dir /s /b "%USERPROFILE%\iniciar_totyshop.bat" 2^>nul') do if not defined PROJ if exist "%%~dpFbackend\server.py" set "PROJ=%%~dpF"
if not defined PROJ echo NAO ACHEI A PASTA DO PROJETO
if not defined PROJ pause
if not defined PROJ exit /b 1
echo Pasta encontrada: %PROJ%
pause
echo Baixando...
powershell -NoProfile -Command "Invoke-WebRequest '%URL%' -OutFile '%ZIP%'"
powershell -NoProfile -Command "Expand-Archive '%ZIP%' '%PROJ%' -Force"
if not exist "%PROJ%backend\dropper_reader.py" echo FALHOU AO INSTALAR
if not exist "%PROJ%backend\dropper_reader.py" pause
if not exist "%PROJ%backend\dropper_reader.py" exit /b 1
echo INSTALADO COM SUCESSO
echo O painel vai ser remontado agora (2 a 5 minutos).
pause
if exist "%PROJ%frontend\build\.stamp-v4" del /q "%PROJ%frontend\build\.stamp-v4"
cd /d "%PROJ%"
call "%PROJ%iniciar_totyshop.bat"
